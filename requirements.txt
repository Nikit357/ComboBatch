# Complete pinned environment for the combobatch Docker image.
#
# This file is the IMAGE's dependency set — it assumes R 4.5.3 and Octave are present.
# For a laptop install use `pip install -e ".[dev]"`, which pulls only the R-free core
# from pyproject.toml. Keep the version constraints here identical to pyproject.toml;
# tests/unit/test_requirements_match_pyproject.py enforces that they do not drift.

# --- core ---
numpy>=1.24,<2.0          # <2.0: rpy2, scanorama and numba all break on the 2.x ABI
pandas>=1.5,<2.0          # <2.0: reComBat 0.1.4 hard requirement, and 1.5.3 has a
                          #       py3.11 wheel only — this is what pins Python to 3.11
scikit-learn>=1.3,<1.6
scipy>=1.9
statsmodels>=0.14
umap-learn>=0.5           # imported as `umap`; needed by metric group C
psutil>=5.9               # memory guard fallback when cgroup files are absent
pyyaml>=6.0

# --- storage ---
boto3>=1.28.17
botocore>=1.31.17
awscli>=1.44              # v1 pins botocore exactly; keep the floor at the pod's value

# --- R bridge ---
rpy2>=3.5.5,<3.7          # ABI-bound to R 4.5.x. rpy2 3.6.x segfaults against R 4.6.0,
                          # which is why the image pins R to 4.5.3.

# --- harmonization methods ---
harmonypy>=0.0.9,<0.2     # 11_harmony. <0.2, not <2.0: 0.2.0 hard-depends on torch,
                          # which drags in nvidia/* and triton — 4.5 GB of CUDA in a
                          # CPU-only image for c6a nodes, and combobatch never imports
                          # torch. 0.1.0 has the same run_harmony API and describes
                          # itself as "NumPy, R package compatible".
scanorama>=1.7            # 12_scanorama
combat>=0.3.3             # 07_pycombat
inmoose>=0.9.1            # 08_inmoose_combatseq
# 30_recombat's reComBat is NOT installed from this file. It declares `sklearn` (the
# deprecated PyPI stub, which now only raises when built) and `python >=3.8,<3.11`, so
# pip refuses it twice over. Both are stale metadata — it runs correctly on 3.11 — so the
# Dockerfile installs it alone with --no-deps --ignore-requires-python, pinned to a
# commit. `--no-deps` is a global pip flag with no per-requirement form, which is the
# whole reason it cannot live here. See docker/recombat_install_plan_260830.md.
tqdm>=4.62.3              # reComBat imports it, and --no-deps means we must declare it

# --- shambhala (20_shambhala) ---
qnorm>=0.4.1              # pulls numba + llvmlite (~56 MB)

# --- test harness ---
# The image runs its own acceptance suite (`pytest tests/integration/
# test_methods_all_backends.py`), which is the only place every backend is actually
# present, so the runner ships with it. Floors match the `dev` extra in pyproject.toml.
pytest>=7.4
pytest-timeout>=2.1
moto[s3]>=4.2

# --- marker panels (metric groups L and M) ---
openpyxl>=3.1             # .xlsx supplementary tables
xlrd>=2.0.1               # legacy .xls (Dybkaer DS1); openpyxl cannot read BIFF8
